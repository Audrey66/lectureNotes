def diff(big: Z, small: Z): Z = {
    Contract(
        Requires( big >= small ),
        Ensures ( Res[Z] == big-small )
    )

    //PREMISES:
    big >= small 


    var count: Z = 0
    var cur: Z = big

    //PREMISES:
    big >= small 
    count == 0
    cur == big


    //SHOW:
    count == (big-cur) //must prove all invariants before loop starts


    while (cur != small) {
        Invariant(
            Modifies(count, cur),
            count == (big-cur)
        )

        //PREMISES:
        big >= small
        count == (big-cur) //list all as premise at start of iteration
        cur != small


        cur = cur - 1
        count = count + 1

        //SHOW:
        count == (big-cur) // must prove all invariants still hold at end of loops


    }

    //PREMISES:
    big >= small
    count == (big-cur) // can list all invariants as premises
    !(cur != small) // loop has ended


    //SHOW:
    count == big-small //prove all postconditions


    return count
}
